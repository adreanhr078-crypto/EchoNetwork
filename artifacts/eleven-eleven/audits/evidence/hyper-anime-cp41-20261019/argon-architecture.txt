	 vec2  p = n.y >  0.65  ? world_point.xz : n.x > n.z ? world_point.zy : world_point.xy;

	 vec2  edge =  min (cell,  1.0  - cell) * panel_pitch;
	 float  min_edge =  min (edge.x, edge.y);
	 float  joint =  1.0  -  smoothstep ( 0.008 ,  0.022 , min_edge);
	 float  lip =  smoothstep ( 0.020 ,  0.032 , min_edge) * ( 1.0  -  smoothstep ( 0.032 ,  0.050 , min_edge));
	 float  stain =  0.5  +  0.5  *  sin (p.x *  1.1 ) *  sin (p.y *  0.85 );

	 vec3  paint = base_color.rgb * ( 0.92  + stain *  0.06  + brush *  0.004 );
	ALBEDO =  mix (paint * ( 1.0  + lip *  0.05 ), paint *  0.78 , joint *  0.65 );
	METALLIC = metal;
	ROUGHNESS =  mix ( clamp (base_roughness + (stain -  0.5 ) *  0.08  - lip *  0.06 ,  0.42 ,  0.82 ),  0.85 , joint *  0.5 );
